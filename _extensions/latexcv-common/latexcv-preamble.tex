% The six-pointed chevrons upstream builds its section headings and bullet

% Pointing left.
\newcommand{\tzlarrow}{(0,0) -- (0.2,0) -- (0.3,0.2) -- (0.2,0.4) -- (0,0.4) -- (0.1,0.2) -- cycle;}
\newcommand{\larrow}[1]{%
  \begin{tikzpicture}[scale=0.58]
    \filldraw[fill=#1!100,draw=#1!100!black] \tzlarrow
  \end{tikzpicture}}

% Pointing right.
\newcommand{\tzrarrow}{(0,0.2) -- (0.1,0) -- (0.3,0) -- (0.2,0.2) -- (0.3,0.4) -- (0.1,0.4) -- cycle;}
\newcommand{\rarrow}[1]{%
  \begin{tikzpicture}[scale=0.7]
    \filldraw[fill=#1!100,draw=#1!100!black] \tzrarrow
  \end{tikzpicture}}

% A run of overlapping chevrons, used to open a section heading.
%
% Upstream writes this inline as `\larrow{c} \hspace{-8pt} \larrow{c}`, where
% the interword spaces either side of the kern are load-bearing: a chevron is
% 0.3cm at scale 0.58, so about 4.95pt wide, and at 10pt the two spaces add back
% about 6.7pt. Each chevron therefore advances roughly 3.6pt.
%
% Written without those spaces the kern is a net -3.05pt, so each chevron lands
% to the *left* of the one before it: the run walks backwards out of the
% heading's coloured box and never displaces the text it is supposed to precede.
% The overlap is set here as an explicit kern so that it states its own size and
% cannot be lost to reflowing whitespace.
\newlength{\cvchevronoverlap}
\setlength{\cvchevronoverlap}{0.8pt}

% \rarrow is drawn at scale 0.7 against \larrow's 0.58, so it is about a fifth
% wider (5.97pt) and takes a proportionally wider kern to overlap by the same
% share of a chevron.
\newlength{\cvchevronroverlap}
\setlength{\cvchevronroverlap}{0.97pt}

\makeatletter
\newcommand{\cv@chevronrun}[4]{%
  \@tempcnta=0\relax
  \@whilenum\@tempcnta<#1\do{%
    \ifnum\@tempcnta>0 \hspace{-#2}\fi
    #3{#4}%
    \advance\@tempcnta by 1\relax}}

% \cvchevrons[count]{colour} -- pointing left, three unless told otherwise.
\newcommand{\cvchevrons}[2][3]{\cv@chevronrun{#1}{\cvchevronoverlap}{\larrow}{#2}}

% \cvchevronsright[count]{colour} -- the same run, pointing right.
\newcommand{\cvchevronsright}[2][3]{\cv@chevronrun{#1}{\cvchevronroverlap}{\rarrow}{#2}}
\makeatother

%-------------------------------------------------------------------------------
%	STAR RATINGS
%-------------------------------------------------------------------------------
% A run of filled stars followed by empty ones, as `sidebar` rates its fields:
% ten \icon{Star} glyphs per line, the first few in the complementary accent
% and the rest in the column's background colour.
%
% Reached from a document through the `stars` shortcode rather than through a
% field on an entry -- `{{< stars 8/10 >}}` in any cell of any listing, or in
% ordinary text. A rating is a way of writing a value, not a kind of value, so
% it belongs to the content and not to the entries interface. It is also why
% the two colours are commands: a listing set on the tinted column of a sidebar
% style needs a different pair from one set on the page, and can say so
% locally without either of them being an option on the document.
\newcommand{\cvstarfullcolor}{sectcol}
\newcommand{\cvstaremptycolor}{softcol}

% Set before the run of stars, so that a design can give a rating a line of its
% own by making this a line break. Empty by default: on the page there is room
% for a rating beside what it rates.
%
% A rating written with nothing before it then opens with an empty line, which
% is a rating of nothing and not worth guarding against -- \lastnodetype cannot
% tell the start of a line from the middle of one here, and the guard fires the
% break at the start of an item either way.
\newcommand{\cvstarsbreak}{}

% Upstream draws both halves with the same solid glyph, distinguished only by
% colour, which reads against a filled column but leaves the empty half of a
% rating invisible on white. The outline face is used for the empty stars here
% so that a rating states its own denominator wherever it is set.
\newcommand{\cvstarfull}{\textcolor{\cvstarfullcolor}{\faIcon{star}}}
\newcommand{\cvstarempty}{\textcolor{\cvstaremptycolor}{\faIcon[regular]{star}}}

\makeatletter
% \cvstars{filled}{total}
\newcommand{\cvstars}[2]{%
  \cvstarsbreak
  \@tempcnta=0\relax
  \@whilenum\@tempcnta<#1\do{\cvstarfull\advance\@tempcnta by 1\relax}%
  \@whilenum\@tempcnta<#2\do{\cvstarempty\advance\@tempcnta by 1\relax}}
\makeatother

%-------------------------------------------------------------------------------
%	RATING BARS
%-------------------------------------------------------------------------------
% A proportion drawn as a filled bar, as upstream's `sidebarleft` rates a skill:
% a track in the soft neutral with the accent filled across the fraction given.
%
% Reached from a document through the `bar` shortcode -- `{{< bar 80% >}}` --
% for the same reason `stars` is a shortcode rather than a field on an entry: a
% rating is a way of writing a value, so it belongs to the content. The two are
% interchangeable, and every style defines both, so a document keeps its ratings
% whichever it uses and whichever design it is rendered with.
%
% Declared here rather than in `sidebarleft` because a bar is not that design's
% property, only its default. \vitaeskillbar there is a caller of this.
\newcommand{\cvbarfullcolor}{sectcol}
\newcommand{\cvbaremptycolor}{softcol}

% How wide a bar is drawn. A macro rather than a length, so that it is expanded
% where the bar is set rather than where it is configured: written in terms of
% \linewidth it then follows the measure it lands in -- the full page in
% `classic`, a table cell in a listing, a narrow column in a sidebar design --
% without a design having to restate it for each.
\newcommand{\cvbarwidth}{0.4\linewidth}
\newcommand{\cvbarheight}{0.15}

% As \cvstarsbreak: empty by default, so a bar sits on the line beside what it
% rates, and set to a line break by a design too narrow for that.
\newcommand{\cvbarbreak}{}

% Resolved into a length before use, because the filled half is written as a
% multiple of the width -- `0.8\cvbarlength` -- and a factor cannot multiply
% another factor, which is what `0.8\cvbarwidth` would come to once \cvbarwidth
% had expanded to `0.4\linewidth`.
\newlength{\cvbarlength}

% \cvbar{fraction} -- #1 between 0 and 1.
%
% `baseline` anchors the picture on the text baseline rather than centring the
% box on it, so a bar written mid-sentence sits on the line rather than
% straddling it and opening the leading above.
\newcommand{\cvbar}[1]{%
  \cvbarbreak
  \setlength{\cvbarlength}{\cvbarwidth}%
  \begin{tikzpicture}[baseline, rounded corners=2pt, very thin]
    \fill [\cvbaremptycolor] (0,0) rectangle (\cvbarlength, \cvbarheight);
    \fill [\cvbarfullcolor] (0,0) rectangle (#1\cvbarlength, \cvbarheight);
  \end{tikzpicture}}

% Gives an otherwise empty box the height of a line, so a \colorbox around it
% is not set flush against the text it contains.
\newcommand{\mystrut}{\rule[-.3\baselineskip]{0pt}{\baselineskip}}

% Vertically centres a box on the current line, so an icon sits level with the

\newcommand{\vcenteredhbox}[1]{\begingroup\setbox0=\hbox{#1}\parbox{\wd0}{\box0}\endgroup}

%-------------------------------------------------------------------------------
%	MEASURED COLUMNS
%-------------------------------------------------------------------------------
% An entry headline and a skills listing both set one or two fixed-width columns
% beside a flexible one: a date column, an institution column, a category
% column. How wide to make them is what this section answers.
%
% Upstream answers it with a constant -- 13.6cm for classic's date, `p{2.4cm}`
% in modern's metasection, `p{2cm}` in rows', `p{0.18\linewidth}` in
% two_column's -- and every one of those holds only because the content beside
% it is hand-written to fit. Content out of a document is arbitrary, and a
% constant fails on it in a way that is easy to miss: a p-column wraps at
% spaces, so a value too wide to break does not overflow anywhere visible, it
% runs on horizontally and overprints the column beside it.
%
% So the columns are measured. Each listing template runs \cvmeasure over its
% entries before setting any of them, growing a length to fit the widest value;
% every row is then set to that width, so the columns line up and are never
% wider than their content needs.
%
% This is why there are no `when-width`, `with-width` or `skill-width` options.
% A fixed length cannot be right for both a full page in `classic` and a narrow
% sidebar column in `sidebar`, and once the width is taken from the content
% there is nothing left for such an option to correct.
%
% What is left to choose is how far a measurement should reach, which is
% `column-widths` in the front matter:
%
%   document  (default) One width per column for the whole document, taken from
%             the widest value anywhere in it, so that entry titles share a left
%             edge across every section. This is the look upstream got from its
%             single fixed 2.3cm, and it needs a round trip: the widths are
%             written to the .aux at the end of a run and read back at the start
%             of the next. LaTeX is run twice as a matter of course, so this
%             settles on its own; a deliberately single-pass render shows the
%             first-pass widths and is warned about below.
%
%   listing   Each listing sizes its own columns. Tighter, and right on a single
%             pass, at the cost of a section of short dates setting its titles
%             further left than a section of long ones.
$if(column-widths)$
\def\cvcolscope{$column-widths$}
$else$
\def\cvcolscope{document}
$endif$
\makeatletter
\def\cv@scopelisting{listing}
\newif\ifcvcoldoc
\ifx\cvcolscope\cv@scopelisting\cvcoldocfalse\else\cvcoldoctrue\fi
\makeatother

% The width each column is actually set at.
\newlength{\cvwhenwidth}
\newlength{\cvwithwidth}
\newlength{\cvskillwidth}

\newlength{\cvskillmax}

% What the previous run measured, read back from the .aux.

\newcommand{\cvmeasure}[2]{%

% Caps a settled column at #2, so that a single runaway value cannot crowd out

%-------------------------------------------------------------------------------
% The .aux round trip that document scope runs on.
%
% \cvcolcarried is what the previous run left behind; it is called as the .aux is
% read at \begin{document}, before any listing is set. At the end of this run the
% maxima just measured are written back, and compared with what was carried in --
% a difference means the widths on the page are the old ones, and a rerun will
% settle them, so it is reported the way LaTeX reports any other rerun.
% The assignments must be global. LaTeX reads the .aux inside a group -- the
% \begingroup ... \@input{\jobname.aux} ... \endgroup in \document -- so a local
% \setlength here is undone the moment the file ends, and every width arrives
% back at zero. It is the same reason \newlabel and friends are global.
\newcommand{\cvcolcarried}[3]{%
  \global\cvwhencarry=#1\relax
  \global\cvwithcarry=#2\relax
  \global\cvskillcarry=#3\relax}

\makeatletter
\AtEndDocument{%
  \ifcvcoldoc
    \immediate\write\@auxout{%
      \string\cvcolcarried{\the\cvwhenmax}{\the\cvwithmax}{\the\cvskillmax}}%
    \ifdim\cvwhenmax=\cvwhencarry
      \ifdim\cvwithmax=\cvwithcarry
        \ifdim\cvskillmax=\cvskillcarry\else\cv@colrerun\fi
      \else\cv@colrerun\fi
    \else\cv@colrerun\fi
  \fi}
\newcommand{\cv@colrerun}{%
  \@latex@warning@no@line{Entry column widths have changed. Rerun to get them
    right, or set `column-widths: listing' to size each listing on its own}}
\makeatother

%-------------------------------------------------------------------------------
%	ENTRY MACROS
%-------------------------------------------------------------------------------
% The listings in `entries/` are shared by every style, so they are written
% against the four commands below rather than against any one style's own
% macros. Each `latexcv-<style>.tex` defines them in its own visual language:
%
%   \cvsection{title}                        a level-1 heading
%   \cvsubsection{title}                     a level-2 heading
%   \vitaeentry{when}{what}{with}{where}{why}   one `detailed` entry
%   \vitaebrief{when}{what}{with}{where}     one `brief` entry
%   \vitaeskill{what}{with}                  one `skills` entry
%   vitaewhy (environment)                   the bullets under a detailed entry
%
% Upstream's own \cvevent is deliberately not carried over. Its argument list
% differs between styles (four arguments in `classic`, five in `modern`, seven
% in `sidebarleft`), and in three of them the elaboration is two fixed
% arguments rather than a list -- so an entry may have exactly two bullets and
% no more. The commands above take the same fields in every style, and take
% the elaboration as a list of any length.
