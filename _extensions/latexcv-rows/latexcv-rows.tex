%-------------------------------------------------------------------------------
%	ROWS
%
%	The page as a stack of full-width bands: a dark title band, then one band
%	per section, shaded alternately so that consecutive sections separate
%	without a rule between them. A section is named not above its band but
%	beside it -- the heading set large and turned on its side in the right
%	margin of the band, behind an accent rule, which is what distinguishes this
%	design from the other six.
%
%	Upstream sizes each band as a minipage of a hand-measured fraction of the
%	text height (0.085, 0.36, 0.2875 ...), which only adds up for the content
%	upstream happens to ship, and cannot break across a page. Here the bands
%	are derived from the document by latexcv.lua -- one per level-1 section --
%	and set as breakable tcolorboxes, so they size themselves and a long
%	section runs onto the next page.
%-------------------------------------------------------------------------------

$if(geometry)$
\geometry{$for(geometry)$$geometry$$sep$,$endfor$}
$else$
% The bands are the page here, so they run from edge to edge of the paper, as
% upstream's do -- its own geometry is left=-0.1cm, right=0cm, which is a
% roundabout way of asking for no side margin at all. A band inset from the
% paper would read as a box on a page rather than as a stripe of it, and the
% stack of them would no longer be the design.
%
% So the page keeps no margin at all -- not at the top or bottom either, where
% a strip of white above the title band or below the footer would read as the
% stack having been placed on the page rather than being it. What holds text
% off the paper's edge is the padding inside each band: \cvbandpad below.
\geometry{top=0cm, bottom=0cm, left=0cm, right=0cm}
$endif$

% `breakable` is what lets a band run onto the next page; `many` brings in the
% skins the breakable library needs.
\usepackage[many]{tcolorbox}
\tcbuselibrary{breakable}

% \needspace, to keep a band from starting so near the foot of a page that the
% piece left there is too short to name. See the cvband wrapper.
\usepackage{needspace}

%-------------------------------------------------------------------------------
%	HEADINGS
%-------------------------------------------------------------------------------
% A heading names the band it introduces from the side, so it is not set where
% it is written. \cvsection only records it, and the band draws it: see the
% overlay below.
%
% Recorded as a box rather than as a token list, so that whatever the heading
% was written with -- emphasis, a link, an inline formula -- is typeset once,
% here, in the same conditions as any other text. \global, because the heading
% arrives outside the band's group.
\newsavebox{\cvbandtitlebox}
\newcommand{\cvsection}[1]{\global\sbox{\cvbandtitlebox}{\HUGE\textsc{#1}}}

% Forget the recorded heading, so that the next band is drawn without one. A
% section split across two bands by an interrupt row is still one section, and
% naming it twice down the page would say it was two. Emitted by latexcv.lua
% between the interrupt row and the band that resumes the section.
\newcommand{\cvbandnotitle}{\global\sbox{\cvbandtitlebox}{}}

\newcommand{\cvsubsection}[1]{%
  \vspace{4pt}
  {\larrow{bgcol}\hspace{2pt}\textcolor{sectcol}{\textbf{#1}}}\par
  \vspace{2pt}}

% A label and its value on one line, as upstream heads its meta block: a
% chevron, the label in the accent colour, and the value beside it. The skills
% listing is set this way -- see \vitaeskill below.
\newcommand{\metasection}[2]{%
  \noindent
  \begin{tabular*}{\linewidth}{@{}p{\dimexpr\cvskillwidth+1em\relax}@{}p{\dimexpr\linewidth-\cvskillwidth-1em\relax}@{}}
    \larrow{bgcol}\hspace{2pt}\textcolor{sectcol}{#1} & #2
  \end{tabular*}\par\vspace{8pt}}

%-------------------------------------------------------------------------------
%	BANDS
%-------------------------------------------------------------------------------
% Where everything across a band sits, measured from the edge of the paper --
% which, since the page has no margins of its own, is also the edge of the
% band.
%
% Upstream arrives at these by stepping across its minipage: \hspace{26pt}, a
% parbox of 0.75\linewidth, \hspace{18pt}, a 2pt rule, \hspace{12pt}, then the
% turned heading. That sequence cannot be reproduced by stepping here, because
% the heading is drawn over the band rather than set beside it in the same
% line -- it has to be placed against an anchor instead. So the sequence was
% measured off upstream's own output at 21cm and is given here as the positions
% it produces:
%
%   text from   1.12cm        text to  16.91cm  (= 4.09cm from the right edge)
%   rule at    17.80cm    heading at   18.79cm
%
% Which is why these are absolute rather than a padding plus an offset: they
% are four points on one line, and it is the distances between them that are
% upstream's design.
\newcommand{\cvbandpad}{1.12cm}

% How much of the band's width is kept clear on the right: the accent rule, the
% turned heading, and the gaps around them. The heading is turned, so what it
% costs horizontally is the height of the face it is set in and not the length
% of the word.
\newlength{\cvbandtitlewidth}
\setlength{\cvbandtitlewidth}{4.09cm}

% The rule and the middle of the heading, leftwards from the edge of the band.
\newlength{\cvbandruleoffset}
\newlength{\cvbandtitleoffset}
\setlength{\cvbandruleoffset}{3.2cm}
\setlength{\cvbandtitleoffset}{2.21cm}
\newcommand{\cvbandrulewidth}{2pt}

% How much of the band's height the accent rule spans.
\newcommand{\cvbandrulefraction}{0.8}

\newlength{\cvbandheight}
\newlength{\cvbandrulelen}

% The heading, turned to read downwards. Always at the size \cvsection set it
% in -- a heading is not a measurement of the section under it, and a page
% whose headings each came out at a different size because of how much each
% section happened to hold would read as an accident. The band is grown to fit
% the heading instead; see \cvbandminheight.
\newcommand{\cvbandtitleset}{%
  \rotatebox[origin=c]{270}{\usebox{\cvbandtitlebox}}}

% The height a band must reach for its heading to sit inside the accent rule,
% as upstream's headings do -- its rule is a fixed 7cm and every heading it
% sets is shorter. Turned, the heading is as tall as it is wide, and the rule
% is \cvbandrulefraction of the band, so the band has to be the heading's width
% over that fraction. 1.25 is the reciprocal of 0.8; the two are written out
% because \dimexpr cannot divide by a decimal.
%
% A band with no heading recorded -- the far side of an interrupt row -- keeps
% no floor at all: the floor is here to make room for a heading, and that band
% has none to make room for.
\newlength{\cvbandminheight}
\newcommand{\cvbandsetminheight}{%
  \ifdim\wd\cvbandtitlebox>0pt\relax
    \setlength{\cvbandminheight}{1.25\wd\cvbandtitlebox}%
    % A floor under the floor, so that a one-syllable heading over a section of
    % two entries still gets a band deep enough to read as one.
    \ifdim\cvbandminheight<3.4cm\relax\setlength{\cvbandminheight}{3.4cm}\fi
  \else
    \setlength{\cvbandminheight}{0pt}%
  \fi}

% Drawn over the band rather than set inside it, because the height to centre
% the heading on is not known until the band has been broken and its pieces
% placed -- which is after its content has been set. The band's own `interior`
% node carries that height, so it is measured here and the rule and heading
% follow from it.
%
% \pgfpointdiff over the two anchors rather than pgf's `let` syntax, so the
% height arrives as a length rather than as a pgf coordinate.
%
% The rule belongs to the heading rather than to the band, so a band with no
% heading recorded -- the second half of a section split by an interrupt row --
% gets neither.
% The overlay runs once per *fragment* of a broken band, not once per band, and
% the fragment a page break leaves at the foot of a page can be a few
% millimetres tall. The heading no longer shrinks to whatever it is given, so a
% fragment too short to hold it is left bare and the heading falls to the next
% one -- where the section continues, and where there is room to read it. Left
% to draw, it would run past the fragment and over the section above.
\newcommand{\cvbandoverlay}{%
  \ifdim\wd\cvbandtitlebox>0pt\relax
  \pgfpointdiff{\pgfpointanchor{interior}{south east}}%
               {\pgfpointanchor{interior}{north east}}%
  \pgfgetlastxy{\cvbandtmpx}{\cvbandtmpy}%
  \setlength{\cvbandheight}{\cvbandtmpy}%
  \setlength{\cvbandrulelen}{\cvbandrulefraction\cvbandheight}%
  \ifdim\cvbandrulelen<\wd\cvbandtitlebox\relax\else
    \draw[sectcol, line width=\cvbandrulewidth]
      ([xshift=-\cvbandruleoffset, yshift=0.5\cvbandrulelen]interior.east) --
      ([xshift=-\cvbandruleoffset, yshift=-0.5\cvbandrulelen]interior.east);
    \node[anchor=center, inner sep=0pt, xshift=-\cvbandtitleoffset]
      at (interior.east) {\cvbandtitleset};%
  \fi
  \fi}

% One shaded band. The colour is passed in by latexcv.lua, which alternates
% `lightcol` and `white` down the document.
%
% `enhanced` is what makes `overlay` draw: under tcolorbox's default skin the
% key is accepted and then ignored, and the heading would go missing from every
% band that happened to fit on one page.
% `before upper` records the tint the band was given, so that what is drawn
% inside it can be mixed into it: the hairline under an entry headline is the
% one thing here that has to read against the band rather than beside it. See
% \cventryrule.
% #2 is the floor under the band's height, worked out by \cvbandsetminheight
% from the heading the band has to hold. It is passed in already expanded to a
% literal -- see the `cvband' wrapper -- because tcolorbox reads `height from'
% with a delimited scan for the `to' between the two bounds, which a macro
% standing for either bound runs past. The upper bound is only a bound, and is
% set past any page so that it never decides where a band breaks.
\newtcolorbox{cvbandbox}[2]{%
  enhanced, breakable,
  colback=#1, colframe=#1,
  before upper={\colorlet{cvbandcol}{#1}},
  boxrule=0pt, arc=0pt, outer arc=0pt, boxsep=0pt,
  left=\cvbandpad, right=\cvbandtitlewidth, top=10pt, bottom=10pt,
  % Bands are stacked flush, as upstream stacks its minipages: with no rule
  % between them, the change of tint is what separates one section from the
  % next, and a gap would break the stripe rather than reinforce it. What looks
  % like a gap between two tinted bands is the white band between them.
  before skip=0pt, after skip=0pt,
  width=\linewidth,
  height from=#2 to 500cm,
  overlay={\cvbandoverlay}}

% One shaded band, sized to hold its heading. The floor has to be worked out
% before the box is opened, because tcolorbox reads the height keys as it opens
% it -- hence the wrapper rather than a key on the box itself.
%
% The same floor is asked of the page before the band starts. A band is free to
% break across pages, but not to begin a few millimetres from the foot of one:
% the heading is drawn per fragment and is no longer shrunk to fit, so a
% fragment shorter than the heading is left bare -- and a band broken into two
% such fragments would go unnamed altogether. \needspace turns that into a page
% break before the band instead, so the piece the heading is drawn on always
% has room for it.
\newenvironment{cvband}[1]{%
  \cvbandsetminheight
  \par
  \ifdim\cvbandminheight>0pt\relax\needspace{\cvbandminheight}\fi
  \begin{cvbandbox}{#1}{\the\cvbandminheight}%
}{%
  \end{cvbandbox}%
}

%-------------------------------------------------------------------------------
%	INTRO BAND
%-------------------------------------------------------------------------------
% The band above the first section: the photograph on the left, and beside it
% whatever the document says before its first heading. Upstream fills this band
% with a photo and a block of \metasection lines; the lines are content here,
% written as ordinary Markdown before the first `#', so what goes beside the
% photo is whatever the document put there.
%
% A band rather than loose content, because the page has no margins: text left
% outside a band would start on the paper's edge. It carries no heading, so it
% reserves nothing on the right and has no floor under its height -- it is as
% tall as the photo or the text beside it, whichever is deeper.
\newtcolorbox{cvintrobox}{%
  enhanced, breakable,
  colback=white, colframe=white,
  before upper={\colorlet{cvbandcol}{white}},
  boxrule=0pt, arc=0pt, outer arc=0pt, boxsep=0pt,
  left=\cvbandpad, right=\cvbandpad, top=10pt,
  % A breakable box takes a \parskip off its own foot, so the band that follows
  % this one starts that much above where the box was drawn to end and paints
  % over the difference. Between two tinted bands that is invisible -- one tint
  % replaces another. Here it is not: this band is white and holds a photograph
  % with a hard edge, so what it eats is the white under the picture, leaving it
  % sitting nearer the band below than the one above. The \parskip is added back
  % so the photograph keeps the same 10pt on both sides.
  bottom=\dimexpr10pt+\parskip\relax,
  before skip=0pt, after skip=0pt,
  width=\linewidth}

\newlength{\cvintropicwidth}
\newlength{\cvintrotextwidth}
% Upstream's \hspace{8pt} between the photo and what stands beside it.
\newcommand{\cvintrogap}{8pt}

% The widths are taken inside the box, where \linewidth is the band's own
% interior rather than the page: 0.2 of it is upstream's `width=0.2\linewidth',
% measured against the parbox it sets its photo in.
%
% Both columns are \vspace{0pt} at the top of a `[t]' minipage, which is what
% aligns them by their first line rather than by their first baseline -- without
% it a photo and a paragraph beside it sit a line apart.
\newenvironment{cvintro}{%
  \begin{cvintrobox}%
$if(profilepic)$
  \setlength{\cvintropicwidth}{$if(profilepic-width)$$profilepic-width$$else$0.2\linewidth$endif$}%
  \setlength{\cvintrotextwidth}{\dimexpr\linewidth-\cvintropicwidth-\cvintrogap\relax}%
  \begin{minipage}[t]{\cvintropicwidth}%
    \vspace{0pt}%
    \includegraphics[width=\linewidth]{$profilepic$}%
  \end{minipage}%
  \hspace{\cvintrogap}%
  \begin{minipage}[t]{\cvintrotextwidth}%
    \vspace{0pt}%
$endif$
}{%
$if(profilepic)$
  \end{minipage}%
$endif$
  \end{cvintrobox}%
}

% The same band without the photograph, for a second run of intro material --
% an interrupt row may split what comes before the first heading, and only the
% first of the resulting bands carries the photo.
\newenvironment{cvintroplain}{\begin{cvintrobox}}{\end{cvintrobox}}

%-------------------------------------------------------------------------------
%	INTERRUPT ROWS
%-------------------------------------------------------------------------------
% A band that carries one line rather than a section: upstream runs two of them
% down its page -- a filled strip under the title holding a summary, and a
% second between two sections pointing at a link. Neither is a section, so
% neither is written as a heading; they are what `::: {.interrupt}` produces.
%
% Shallow, centred, and unbreakable: an interrupt row that ran onto the next
% page would be a section, which is what it is not.
%
% #1 the fill; #2 the colour of the chevrons around the line, or empty for
% none. Upstream's accent strip is flanked by chevrons and its dark one is not,
% and both are wanted, so the marks are a parameter rather than a fixture.
\newtcolorbox{cvinterrupt}[2]{%
  enhanced, breakable=false,
  colback=#1, colframe=#1,
  boxrule=0pt, arc=0pt, outer arc=0pt, boxsep=0pt,
  left=\cvbandpad, right=\cvbandpad, top=6pt, bottom=6pt,
  before skip=0pt, after skip=0pt,
  width=\linewidth,
  halign=center,
  fontupper=\large\color{white},
  before upper={\ifthenelse{\isempty{#2}}{}{\larrow{#2}\larrow{#2}\hspace{4pt}}},
  after upper={\ifthenelse{\isempty{#2}}{}{\hspace{4pt}\rarrow{#2}\rarrow{#2}}}}

%-------------------------------------------------------------------------------
%	ENTRIES
%-------------------------------------------------------------------------------
% The institution and the location, set together in brackets after the title of
% an entry, in the dark neutral. Upstream's `rows` writes this as a plain
% `(#3)`; it is a command here because either field may be absent, and an entry
% missing one should not print the brackets around nothing.
\newcommand{\cventrywith}[2]{%
  \ifthenelse{\isempty{#1}}{%
    \ifthenelse{\isempty{#2}}{}{\ \textcolor{bgcol}{(#2)}}%
  }{%
    \ifthenelse{\isempty{#2}}{\ \textcolor{bgcol}{(#1)}}%
                             {\ \textcolor{bgcol}{(#1, #2)}}%
  }}

% Title and institution from the left, date from the right, as one line. The
% date column is measured across the listing (see MEASURED COLUMNS in
% latexcv-preamble.tex) so the dates share a right edge without the title
% column being cut to a guess.
\newcommand{\cventryhead}[4]{%
  \begin{tabular*}{\linewidth}{@{}p{\dimexpr\linewidth-\cvwhenwidth\relax}@{}x{\cvwhenwidth}@{}}
    \textbf{#2}\cventrywith{#3}{#4} & \textcolor{bgcol}{#1}
  \end{tabular*}}

% The hairline under an entry headline, drawn against whichever band it landed
% on. Upstream sets it in a fixed pale grey, which it could: it has two tints
% and chose a grey that sits between them. Here a band may be either, so a
% fixed grey is wrong on one of them -- against white it is barely a line, and
% against the tint it is *lighter* than what it is drawn on, so it reads as a
% gap rather than as a rule. Mixed into the band's own colour instead, so the
% contrast is the same wherever the entry falls.
%
% \color rather than \textcolor: \textcolor is for text, and begins a paragraph
% before it sets the colour, which leaves the rule a line's leading lower than
% the vertical space here asks for. That is what was cutting the descenders of
% the headline above it.
\colorlet{cvbandcol}{white}
\newcommand{\cventryrule}{{\color{bgcol!30!cvbandcol}\hrule}}

% Upstream's \cvevent, step for step: 8pt above the headline, the headline, the
% rule pulled up under it, 6pt, then the elaboration. The vertical measurements
% are upstream's own -- an entry that is spaced more tightly than upstream's
% reads as a denser document, which is a different design and not this one.
\newcommand{\vitaeentry}[5]{%
  \vspace{8pt}
  \cventryhead{#1}{#2}{#3}{#4}
  % Upstream ends its headline row with \\[4pt] before pulling the rule up by
  % 12pt; the row here ends without one, so the same clearance is 8pt.
  \vspace{-8pt}
  \cventryrule
  \vspace{6pt}
  #5}

\newcommand{\vitaebrief}[4]{%
  \vspace{3pt}
  \cventryhead{#1}{#2}{#3}{#4}}

% A skills listing is upstream's meta block: this design has no second style
% for a label and a value, and does not need one.
\newcommand{\vitaeskill}[2]{\metasection{#1}{#2}}

% Flush with the entry above rather than indented under its title: upstream
% sets its two elaboration lines in a one-column tabular at the entry's own left
% edge, and the entries here are separated by a rule, so the bullets have no
% column to hang from either. `leftmargin` is the whole of the indent an
% itemize would otherwise add, so it is taken back to nothing and the chevron
% left to start the line.
%
% 4.5pt between bullets is upstream's \\[4.5pt]. The 4pt under the last of them
% is measured rather than copied: upstream writes \\[6pt] and then takes 4pt
% back, but it is closing a tabular, whose last row carries a depth that an
% itemize's last line does not, and 4pt is what puts the next entry's headline
% where upstream puts it.
% `leftmargin=*` is what puts the chevron itself on the entry's left edge:
% enumitem then takes the indent from the label's own width, so the label
% begins at nothing and the text hangs off it. Setting `leftmargin` to zero
% instead would put the *text* on the edge and hang the chevron out past it,
% into the band's margin.
\newenvironment{vitaewhy}{%
  \begin{itemize}[label=\larrow{sectcol}, topsep=0pt, partopsep=0pt,
                  itemsep=4.5pt, parsep=0pt, leftmargin=*, labelsep=0.4em,
                  align=left]%
}{%
  \end{itemize}%
  \vspace{4pt}%
}
